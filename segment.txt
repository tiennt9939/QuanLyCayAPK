async function refreshOrdersForNotification(){
  const st=window.CloudSync?.state?.();
  if(!st?.user||!db)return [];
  const watched=Array.isArray(orderNotificationState.rows)?orderNotificationState.rows.filter(Boolean):[];
  const cached=Array.isArray(D.packingOrders)?D.packingOrders.filter(Boolean):[];
  const filterMine=rows=>st.role==='packer'
    ? rows.filter(x=>String(x.assignedUid||'')===String(st.user.uid))
    : rows;
  try{
    // Khi người dùng CHỦ ĐỘNG bấm chuông, đọc thẳng Firestore một lần.
    // Đây là nguồn xác nhận cuối cùng; listener chỉ dùng để cập nhật badge theo thời gian thực.
    // Với packer, query luôn phải có assignedUid == auth.uid để khớp Security Rules.
    let fresh=[];
    if(st.role==='packer'){
      const snap=await db.collection('packingOrders').where('assignedUid','==',st.user.uid).get();
      fresh=normalizeOrders(snap.docs.map(d=>({id:d.id,...d.data()})));
    }else if(st.role==='admin'){
      const snap=await db.collection('packingOrders').get();
      fresh=normalizeOrders(snap.docs.map(d=>({id:d.id,...d.data()})));
    }
    const freshMine=filterMine(fresh);
    // Với nhân viên, listener đã chứng minh có đơn thì một lần get() trả []
    // không được phép xóa danh sách đang hiển thị. Đây là điểm gây lỗi:
    // vừa đăng nhập thấy 8 đơn, bấm XEM ĐƠN NGAY lại thành 0 đơn.
    // Listener là nguồn realtime; get() chỉ dùng để làm mới khi có dữ liệu.
    let rows=freshMine;
    const known=filterMine(watched.length?watched:cached);
    if(st.role==='packer' && !freshMine.length && known.length){
      rows=known;
    }else if(!freshMine.length && fresh.length===0 && known.length){
      rows=known;
    }else if(!freshMine.length && fresh.length>0){
      rows=known;
    }
    orderNotificationState.rows=rows.slice();
    if(st.role==='packer')D.packingOrders=rows.slice();
    else if(st.role==='admin')D.packingOrders=rows.slice();
    localSaveOnly();
    if(st.role==='admin')await renderAssignmentControls();
    const pending=rows.filter(x=>x.status!=='completed').length;
    const completed=rows.filter(x=>x.status==='completed').length;
    updateOrderNotificationUI(pending,completed);
    return rows;
  }catch(e){
    console.error('refreshOrdersForNotification',e);
    // Không biến lỗi mạng/quyền thành “0 đơn”. Giữ dữ liệu listener/cache nếu còn.
    let rows=filterMine(watched.length?watched:cached);
    if(!rows.length && st.role==='packer')rows=restorePackerOrderCache(st.user.uid);
    if(rows.length){
      updateOrderNotificationUI(rows.filter(x=>x.status!=='completed').length,rows.filter(x=>x.status==='completed').length);
      return rows;
    }
    toast('Không tải được đơn: '+(e?.message||e?.code||'Lỗi Cloud'));
    return [];
  }
}

async function openOrderNotifications(){
  // V24 ROOT FIX: the bell must NEVER be blocked by the transient Firebase Auth session gate.
  // If the realtime listener already populated the bell (badge > 0), those rows are the
  // authenticated user's canonical /packingOrders data. Open them immediately.
  unlockOrderAlertAudio();
  try{
    let st=window.CloudSync?.state?.()||{};
    const bellRole=String(orderNotificationState.role||st.role||'');
    let rows=Array.isArray(orderNotificationState.rows)?orderNotificationState.rows.filter(Boolean):[];
    if(!rows.length)rows=Array.isArray(D.packingOrders)?D.packingOrders.filter(Boolean):[];

    // Listener-owned rows are already permission-scoped: packer rows were queried by assignedUid.
    // Only filter when a concrete UID is available; never discard valid listener rows because
    // firebase.auth().currentUser is temporarily null inside Android WebView.
    const uid=String(st?.user?.uid||'');
    if(bellRole==='packer'&&uid)rows=rows.filter(x=>String(x.assignedUid||'')===uid);

    const pending=rows.filter(x=>x&&x.status!=='completed');
    if(pending.length){
      orderNotificationState.rows=rows.slice();
      if(bellRole==='packer'&&uid)D.packingOrders=rows.slice();
      showOrderAlert(pending.length,bellRole==='admin'?'admin':'packer',rows);
      return;
    }

    // No cached rows: only now try a Firestore refresh. This is a data-load fallback,
    // not an Auth gate for the bell.
    let fresh=[];
    try{fresh=await refreshOrdersForNotification()}catch(e){console.warn('bell refresh',e)}
    const freshPending=(Array.isArray(fresh)?fresh:[]).filter(x=>x&&x.status!=='completed');
    if(freshPending.length){
      showOrderAlert(freshPending.length,bellRole==='admin'?'admin':'packer',fresh);
      return;
    }
    toast('🔔 Hiện không có đơn nào đang chờ xử lý');
  }catch(e){
    console.error('openOrderNotifications',e);
    // Never show the old misleading "Firebase session" error from the bell.
    toast('Không thể mở danh sách đơn. Dữ liệu đơn vẫn được giữ nguyên.');
  }
}
